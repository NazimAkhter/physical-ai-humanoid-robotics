## Reinforcement Learning Fundamentals for Robotics

### RL Framework in Robotics

Reinforcement learning in robotics involves an agent (robot) interacting with an environment to learn optimal behaviors:

```
State (s) → Action (a) → Reward (r) → Next State (s')
```

In robotics, this translates to:
- **State**: Sensor readings, robot configuration, environment observations
- **Action**: Motor commands, joint velocities, navigation commands
- **Reward**: Task completion, efficiency, safety metrics
- **Environment**: Physical or simulated world

### RL Problem Formulation

The goal is to learn a policy π(a|s) that maximizes expected cumulative reward:

```
J(π) = E[Σ γ^t * r_t | π]
```

Where γ is the discount factor and the expectation is over trajectories generated by following policy π.

## Deep Reinforcement Learning Algorithms

### Deep Q-Network (DQN) for Continuous Action Spaces

```python
import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
import numpy as np
import random
from collections import deque
import copy

class DQN(nn.Module):
    """Deep Q-Network for discrete action spaces"""
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(DQN, self).__init__()

        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )

    def forward(self, state):
        return self.network(state)

class Actor(nn.Module):
    """Actor network for continuous action spaces (DDPG)"""
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(Actor, self).__init__()

        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh()  # Actions are in [-1, 1]
        )

    def forward(self, state):
        return self.network(state)

class Critic(nn.Module):
    """Critic network for continuous action spaces (DDPG)"""
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(Critic, self).__init__()

        self.state_net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim // 2),
            nn.ReLU()
        )

        self.action_net = nn.Sequential(
            nn.Linear(action_dim, hidden_dim // 2),
            nn.ReLU()
        )

        self.value_net = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )

    def forward(self, state, action):
        state_embedding = self.state_net(state)
        action_embedding = self.action_net(action)

        concat = torch.cat([state_embedding, action_embedding], dim=1)
        value = self.value_net(concat)

        return value

class ReplayBuffer:
    """Experience replay buffer for DQN and DDPG"""
    def __init__(self, capacity):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        state, action, reward, next_state, done = map(np.stack, zip(*batch))
        return state, action, reward, next_state, done

    def __len__(self):
        return len(self.buffer)

class DDPGAgent:
    """Deep Deterministic Policy Gradient agent"""
    def __init__(self, state_dim, action_dim, lr_actor=1e-4, lr_critic=1e-3,
                 gamma=0.99, tau=1e-3, buffer_size=1000000):

        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        # Networks
        self.actor = Actor(state_dim, action_dim).to(self.device)
        self.actor_target = Actor(state_dim, action_dim).to(self.device)
        self.critic = Critic(state_dim, action_dim).to(self.device)
        self.critic_target = Critic(state_dim, action_dim).to(self.device)

        # Copy parameters to target networks
        self.actor_target.load_state_dict(self.actor.state_dict())
        self.critic_target.load_state_dict(self.critic.state_dict())

        # Optimizers
        self.actor_optimizer = optim.Adam(self.actor.parameters(), lr=lr_actor)
        self.critic_optimizer = optim.Adam(self.critic.parameters(), lr=lr_critic)

        # Hyperparameters
        self.gamma = gamma  # Discount factor
        self.tau = tau      # Soft update parameter
        self.noise_std = 0.2  # Exploration noise

        # Replay buffer
        self.replay_buffer = ReplayBuffer(buffer_size)

    def get_action(self, state, add_noise=True):
        """Get action from policy"""
        state = torch.FloatTensor(state).unsqueeze(0).to(self.device)
        action = self.actor(state).cpu().data.numpy().flatten()

        if add_noise:
            noise = np.random.normal(0, self.noise_std, size=action.shape)
            action = action + noise

        return np.clip(action, -1, 1)  # Clamp to valid range

    def update(self, batch_size=100):
        """Update the networks using a batch of experiences"""
        if len(self.replay_buffer) < batch_size:
            return

        # Sample batch from replay buffer
        state, action, reward, next_state, done = self.replay_buffer.sample(batch_size)

        state = torch.FloatTensor(state).to(self.device)
        action = torch.FloatTensor(action).to(self.device)
        reward = torch.FloatTensor(reward).unsqueeze(1).to(self.device)
        next_state = torch.FloatTensor(next_state).to(self.device)
        done = torch.BoolTensor(done).unsqueeze(1).to(self.device)

        # Critic update
        next_action = self.actor_target(next_state)
        next_q_values = self.critic_target(next_state, next_action)
        target_q_values = reward + (self.gamma * next_q_values * ~done)

        current_q_values = self.critic(state, action)

        critic_loss = F.mse_loss(current_q_values, target_q_values.detach())

        self.critic_optimizer.zero_grad()
        critic_loss.backward()
        self.critic_optimizer.step()

        # Actor update
        actor_loss = -self.critic(state, self.actor(state)).mean()

        self.actor_optimizer.zero_grad()
        actor_loss.backward()
        self.actor_optimizer.step()

        # Soft update target networks
        self.soft_update(self.actor, self.actor_target)
        self.soft_update(self.critic, self.critic_target)

    def soft_update(self, local_model, target_model):
        """Soft update model parameters"""
        for target_param, local_param in zip(target_model.parameters(), local_model.parameters()):
            target_param.data.copy_(self.tau * local_param.data + (1.0 - self.tau) * target_param.data)

class PPOAgent:
    """Proximal Policy Optimization agent"""
    def __init__(self, state_dim, action_dim, lr=3e-4, gamma=0.99,
                 eps_clip=0.2, K_epochs=4, device='cpu'):

        self.device = device
        self.gamma = gamma
        self.eps_clip = eps_clip
        self.K_epochs = K_epochs

        # Actor-Critic network
        self.policy = ActorCritic(state_dim, action_dim).to(self.device)
        self.optimizer = optim.Adam(self.policy.parameters(), lr=lr)

        self.policy_old = ActorCritic(state_dim, action_dim).to(self.device)
        self.policy_old.load_state_dict(self.policy.state_dict())

        self.MseLoss = nn.MSELoss()

    def update(self, memory):
        # Monte Carlo estimate of state rewards
        rewards = []
        discounted_reward = 0
        for reward, is_terminal in zip(reversed(memory.rewards), reversed(memory.is_terminals)):
            if is_terminal:
                discounted_reward = 0
            discounted_reward = reward + (self.gamma * discounted_reward)
            rewards.insert(0, discounted_reward)

        # Normalizing the rewards
        rewards = torch.tensor(rewards, dtype=torch.float32).to(self.device)
        rewards = (rewards - rewards.mean()) / (rewards.std() + 1e-5)

        # Convert list to tensor
        old_states = torch.squeeze(torch.stack(memory.states)).to(self.device).detach()
        old_actions = torch.squeeze(torch.stack(memory.actions)).to(self.device).detach()
        old_logprobs = torch.squeeze(torch.stack(memory.logprobs)).to(self.device).detach()

        # Optimize policy for K epochs
        for _ in range(self.K_epochs):
            # Evaluating old actions and values
            logprobs, state_values, dist_entropy = self.policy.evaluate(old_states, old_actions)

            # Finding the ratio (pi_theta / pi_theta__old)
            ratios = torch.exp(logprobs - old_logprobs.detach())

            # Finding Surrogate Loss
            advantages = rewards - state_values.detach()
            surr1 = ratios * advantages
            surr2 = torch.clamp(ratios, 1-self.eps_clip, 1+self.eps_clip) * advantages

            # Final loss of clipped objective PPO
            loss = -torch.min(surr1, surr2) + 0.5 * self.MseLoss(state_values, rewards) - 0.01 * dist_entropy

            # Take gradient step
            self.optimizer.zero_grad()
            loss.mean().backward()
            self.optimizer.step()

        # Copy new weights into old policy
        self.policy_old.load_state_dict(self.policy.state_dict())

class ActorCritic(nn.Module):
    def __init__(self, state_dim, action_dim, n_latent_var=256):
        super(ActorCritic, self).__init__()

        # Actor
        self.action_layer = nn.Sequential(
            nn.Linear(state_dim, n_latent_var),
            nn.Tanh(),
            nn.Linear(n_latent_var, n_latent_var),
            nn.Tanh(),
            nn.Linear(n_latent_var, action_dim),
            nn.Tanh()
        )

        # Critic
        self.value_layer = nn.Sequential(
            nn.Linear(state_dim, n_latent_var),
            nn.Tanh(),
            nn.Linear(n_latent_var, n_latent_var),
            nn.Tanh(),
            nn.Linear(n_latent_var, 1)
        )

    def forward(self):
        raise NotImplementedError

    def act(self, state):
        action_probs = self.action_layer(state)
        action_probs = F.softmax(action_probs, dim=-1)
        dist = torch.distributions.Categorical(action_probs)

        action = dist.sample()
        action_logprob = dist.log_prob(action)

        return action.detach(), action_logprob.detach()

    def evaluate(self, state, action):
        action_probs = self.action_layer(state)
        action_probs = F.softmax(action_probs, dim=-1)
        dist = torch.distributions.Categorical(action_probs)

        action_logprobs = dist.log_prob(action)
        dist_entropy = dist.entropy()

        state_values = self.value_layer(state)

        return action_logprobs, torch.squeeze(state_values), dist_entropy
```

```python
import torch
import numpy as np
from torch.utils.tensorboard import SummaryWriter
import os

class RLTrainer:
    """Reinforcement Learning trainer for Isaac Sim environments"""

    def __init__(self, agent, env, log_dir="logs/rl_training"):
        self.agent = agent
        self.env = env
        self.log_dir = log_dir
        self.writer = SummaryWriter(log_dir)

        # Training parameters
        self.max_episodes = 10000
        self.max_steps_per_episode = 500
        self.save_interval = 1000
        self.eval_interval = 100

        # Episode tracking
        self.episode_count = 0
        self.total_steps = 0

    def train(self):
        """Main training loop"""
        print("Starting RL training...")

        for episode in range(self.max_episodes):
            self.episode_count = episode

            # Reset environment
            obs = self.env.reset()

            episode_reward = 0
            episode_steps = 0

            # Run episode
            for step in range(self.max_steps_per_episode):
                # Get action from agent
                if hasattr(self.agent, 'get_action'):
                    # DDPG/PPO agent
                    action = self.agent.get_action(obs['obs'])
                else:
                    # Other agent types
                    action = self.select_action(obs['obs'])

                # Take action in environment
                next_obs, reward, done, info = self.env.step(action)

                # Store experience
                if hasattr(self.agent, 'replay_buffer'):
                    self.agent.replay_buffer.push(
                        obs['obs'], action, reward, next_obs['obs'], done
                    )

                # Update agent
                if hasattr(self.agent, 'update'):
                    self.agent.update()

                # Update tracking variables
                obs = next_obs
                episode_reward += reward
                episode_steps += 1
                self.total_steps += 1

                if done:
                    break

            # Log episode results
            self.log_episode(episode, episode_reward, episode_steps, info)

            # Save model periodically
            if episode % self.save_interval == 0:
                self.save_model(episode)

            # Evaluate periodically
            if episode % self.eval_interval == 0:
                self.evaluate(episode)

        print("Training completed!")
        self.writer.close()

    def log_episode(self, episode, reward, steps, info):
        """Log episode information"""
        self.writer.add_scalar('Episode/Reward', reward, episode)
        self.writer.add_scalar('Episode/Steps', steps, episode)
        self.writer.add_scalar('Episode/Steps_per_Second', steps / self.env.rendering_dt, episode)

        if 'success_rate' in info:
            self.writer.add_scalar('Episode/Success_Rate', info['success_rate'], episode)

        print(f"Episode {episode}: Reward = {reward:.2f}, Steps = {steps}")

    def evaluate(self, episode):
        """Evaluate the trained agent"""
        eval_episodes = 10
        total_reward = 0
        success_count = 0

        for eval_ep in range(eval_episodes):
            obs = self.env.reset()
            episode_reward = 0
            done = False

            while not done:
                if hasattr(self.agent, 'get_action'):
                    action = self.agent.get_action(obs['obs'], add_noise=False)
                else:
                    action = self.select_action(obs['obs'])

                obs, reward, done, info = self.env.step(action)
                episode_reward += reward

            total_reward += episode_reward

            # Check for success (based on environment-specific criteria)
            if 'success' in info and info['success']:
                success_count += 1

        avg_reward = total_reward / eval_episodes
        success_rate = success_count / eval_episodes

        self.writer.add_scalar('Evaluation/Average_Reward', avg_reward, episode)
        self.writer.add_scalar('Evaluation/Success_Rate', success_rate, episode)

        print(f"Evaluation at episode {episode}: Avg Reward = {avg_reward:.2f}, Success Rate = {success_rate:.2%}")

    def save_model(self, episode):
        """Save the trained model"""
        save_path = f"{self.log_dir}/model_{episode}.pth"

        if hasattr(self.agent, 'actor'):
            # Save actor and critic networks
            torch.save({
                'actor_state_dict': self.agent.actor.state_dict(),
                'critic_state_dict': self.agent.critic.state_dict(),
                'actor_optimizer_state_dict': self.agent.actor_optimizer.state_dict(),
                'critic_optimizer_state_dict': self.agent.critic_optimizer.state_dict(),
                'episode': episode
            }, save_path)
        else:
            # Save other types of agents
            torch.save(self.agent.state_dict(), save_path)

        print(f"Model saved to {save_path}")
```

```python
class CurriculumLearning:
    """Curriculum learning for RL training"""

    def __init__(self, tasks, difficulty_thresholds):
        self.tasks = tasks  # List of tasks from easy to hard
```

```python
    """Deploy trained model to real robot"""
    # Load trained model
    model = torch.load(model_path)
    model.eval()

    # Optimize for robot hardware
    optimized_model = optimize_model_for_deployment(model)

    # Connect to robot
    robot = robot_interface()

    # Main control loop
    try:
        while True:
            # Get sensor observations from robot
            obs = robot.get_observations()

            # Preprocess observations
            obs_tensor = torch.FloatTensor(obs).unsqueeze(0)

            # Get action from model
            with torch.no_grad():
                action = optimized_model(obs_tensor)

            # Convert action to robot commands
            robot_cmd = process_action_for_robot(action)

            # Send command to robot
            robot.send_command(robot_cmd)

    except KeyboardInterrupt:
        print("Deployment stopped by user")
    finally:
        robot.shutdown()
```
